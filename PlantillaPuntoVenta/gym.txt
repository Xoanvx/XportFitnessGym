    // ======== Modelo ========
    static class Producto {
        int id;                       // 0 = aún no guardado en la BD
        String nombre;
        String codigoBarras;          // opcional, único
        double precio;
        String rutaImagen;
        BufferedImage imagen;
        boolean destacado, enDescuento;
        /** talla -> stock. Si está vacío, el producto no maneja tallas. */
        Map<String, Integer> tallas = new LinkedHashMap<>();

        Producto(String nombre, double precio, String rutaImagen) {
            this.nombre = nombre;
            this.precio = precio;
            setImagen(rutaImagen);
        }

        void setImagen(String ruta) {
            this.rutaImagen = ruta;
            this.imagen = null;
            if (ruta != null) {
                try { imagen = ImageIO.read(new File(ruta)); } catch (Exception e) { imagen = null; }
            }
        }

        boolean tieneTallas() { return !tallas.isEmpty(); }

        int stockTotal() {
            int s = 0;
            for (int v : tallas.values()) s += v;
            return s;
        }
    }

    static class ItemCarrito {
        Producto producto;
        String talla;                 // null si el producto no maneja tallas
        int cantidad = 1;
        ItemCarrito(Producto p, String talla) { this.producto = p; this.talla = talla; }
        double total() { return producto.precio * cantidad; }
        int stockDisponible() {
            return talla == null ? Integer.MAX_VALUE : producto.tallas.getOrDefault(talla, 0);
        }
    }

    static class BD {
        private final Connection con;

        BD(String archivo) throws SQLException {
            con = DriverManager.getConnection("jdbc:sqlite:" + archivo);
            try (Statement st = con.createStatement()) {
                st.execute("PRAGMA foreign_keys = ON");
                st.execute("CREATE TABLE IF NOT EXISTS productos ("
                        + "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                        + "nombre TEXT NOT NULL,"
                        + "precio REAL NOT NULL,"
                        + "ruta_imagen TEXT,"
                        + "destacado INTEGER NOT NULL DEFAULT 0,"
                        + "en_descuento INTEGER NOT NULL DEFAULT 0)");
                // Migración: agrega la columna codigo_barras si la BD es de una versión anterior
                boolean tieneCodigo = false;
                try (ResultSet rs = st.executeQuery("PRAGMA table_info(productos)")) {
                    while (rs.next()) {
                        if ("codigo_barras".equals(rs.getString("name"))) tieneCodigo = true;
                    }
                }
                if (!tieneCodigo) st.execute("ALTER TABLE productos ADD COLUMN codigo_barras TEXT");
                st.execute("CREATE UNIQUE INDEX IF NOT EXISTS idx_codigo_barras ON productos(codigo_barras)");
                st.execute("CREATE TABLE IF NOT EXISTS tallas ("
                        + "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                        + "producto_id INTEGER NOT NULL,"
                        + "talla TEXT NOT NULL,"
                        + "stock INTEGER NOT NULL DEFAULT 0,"
                        + "UNIQUE(producto_id, talla),"
                        + "FOREIGN KEY(producto_id) REFERENCES productos(id) ON DELETE CASCADE)");
                st.execute("CREATE TABLE IF NOT EXISTS ventas ("
                        + "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                        + "fecha TEXT NOT NULL DEFAULT (datetime('now','localtime')),"
                        + "cliente TEXT,"
                        + "metodo TEXT,"
                        + "descuento REAL NOT NULL DEFAULT 0,"
                        + "envio REAL NOT NULL DEFAULT 0,"
                        + "total REAL NOT NULL,"
                        + "nota TEXT)");
                st.execute("CREATE TABLE IF NOT EXISTS venta_items ("
                        + "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                        + "venta_id INTEGER NOT NULL,"
                        + "producto_id INTEGER,"
                        + "nombre TEXT NOT NULL,"
                        + "talla TEXT,"
                        + "cantidad INTEGER NOT NULL,"
                        + "precio REAL NOT NULL,"
                        + "FOREIGN KEY(venta_id) REFERENCES ventas(id) ON DELETE CASCADE)");
            }
        }

        List<Producto> cargarProductos() throws SQLException {
            List<Producto> lista = new ArrayList<>();
            try (Statement st = con.createStatement();
                 ResultSet rs = st.executeQuery(
                         "SELECT id,nombre,precio,ruta_imagen,destacado,en_descuento,codigo_barras FROM productos ORDER BY id")) {
                while (rs.next()) {
                    Producto p = new Producto(rs.getString(2), rs.getDouble(3), rs.getString(4));
                    p.id = rs.getInt(1);
                    p.destacado = rs.getInt(5) == 1;
                    p.enDescuento = rs.getInt(6) == 1;
                    p.codigoBarras = rs.getString(7);
                    lista.add(p);
                }
            }
            try (PreparedStatement ps = con.prepareStatement(
                    "SELECT talla,stock FROM tallas WHERE producto_id=? ORDER BY id")) {
                for (Producto p : lista) {
                    ps.setInt(1, p.id);
                    try (ResultSet rs = ps.executeQuery()) {
                        while (rs.next()) p.tallas.put(rs.getString(1), rs.getInt(2));
                    }
                }
            }
            return lista;
        }

        void insertarProducto(Producto p) throws SQLException {
            con.setAutoCommit(false);
            try {
                try (PreparedStatement ps = con.prepareStatement(
                        "INSERT INTO productos(nombre,precio,ruta_imagen,destacado,en_descuento,codigo_barras) VALUES(?,?,?,?,?,?)",
                        Statement.RETURN_GENERATED_KEYS)) {
                    ps.setString(1, p.nombre);
                    ps.setDouble(2, p.precio);
                    ps.setString(3, p.rutaImagen);
                    ps.setInt(4, p.destacado ? 1 : 0);
                    ps.setInt(5, p.enDescuento ? 1 : 0);
                    ps.setString(6, p.codigoBarras);
                    ps.executeUpdate();
                    try (ResultSet k = ps.getGeneratedKeys()) { if (k.next()) p.id = k.getInt(1); }
                }
                guardarTallas(p);
                con.commit();
            } catch (SQLException e) {
                con.rollback();
                throw e;
            } finally {
                con.setAutoCommit(true);
            }
        }

        void actualizarProducto(Producto p) throws SQLException {
            con.setAutoCommit(false);
            try {
                try (PreparedStatement ps = con.prepareStatement(
                        "UPDATE productos SET nombre=?,precio=?,ruta_imagen=?,destacado=?,en_descuento=?,codigo_barras=? WHERE id=?")) {
                    ps.setString(1, p.nombre);
                    ps.setDouble(2, p.precio);
                    ps.setString(3, p.rutaImagen);
                    ps.setInt(4, p.destacado ? 1 : 0);
                    ps.setInt(5, p.enDescuento ? 1 : 0);
                    ps.setString(6, p.codigoBarras);
                    ps.setInt(7, p.id);
                    ps.executeUpdate();
                }
                guardarTallas(p);
                con.commit();
            } catch (SQLException e) {
                con.rollback();
                throw e;
            } finally {
                con.setAutoCommit(true);
            }
        }

        private void guardarTallas(Producto p) throws SQLException {
            try (PreparedStatement del = con.prepareStatement("DELETE FROM tallas WHERE producto_id=?")) {
                del.setInt(1, p.id);
                del.executeUpdate();
            }
            try (PreparedStatement ins = con.prepareStatement(
                    "INSERT INTO tallas(producto_id,talla,stock) VALUES(?,?,?)")) {
                for (Map.Entry<String, Integer> e : p.tallas.entrySet()) {
                    ins.setInt(1, p.id);
                    ins.setString(2, e.getKey());
                    ins.setInt(3, e.getValue());
                    ins.executeUpdate();
                }
            }
        }

        void eliminarProducto(Producto p) throws SQLException {
            try (PreparedStatement ps = con.prepareStatement("DELETE FROM productos WHERE id=?")) {
                ps.setInt(1, p.id);
                ps.executeUpdate();
            }
        }

        /** Guarda la venta y descuenta el stock de cada talla (todo en una transacción). */
        void registrarVenta(List<ItemCarrito> items, double descuento, double envio, double total,
                            String metodo, String cliente, String nota) throws SQLException {
            con.setAutoCommit(false);
            try {
                int ventaId;
                try (PreparedStatement ps = con.prepareStatement(
                        "INSERT INTO ventas(cliente,metodo,descuento,envio,total,nota) VALUES(?,?,?,?,?,?)",
                        Statement.RETURN_GENERATED_KEYS)) {
                    ps.setString(1, cliente);
                    ps.setString(2, metodo);
                    ps.setDouble(3, descuento);
                    ps.setDouble(4, envio);
                    ps.setDouble(5, total);
                    ps.setString(6, nota);
                    ps.executeUpdate();
                    try (ResultSet k = ps.getGeneratedKeys()) { k.next(); ventaId = k.getInt(1); }
                }
                try (PreparedStatement ins = con.prepareStatement(
                             "INSERT INTO venta_items(venta_id,producto_id,nombre,talla,cantidad,precio) VALUES(?,?,?,?,?,?)");
                     PreparedStatement stock = con.prepareStatement(
                             "UPDATE tallas SET stock = stock - ? WHERE producto_id=? AND talla=? AND stock >= ?")) {
                    for (ItemCarrito it : items) {
                        ins.setInt(1, ventaId);
                        ins.setInt(2, it.producto.id);
                        ins.setString(3, it.producto.nombre);
                        ins.setString(4, it.talla);
                        ins.setInt(5, it.cantidad);
                        ins.setDouble(6, it.producto.precio);
                        ins.executeUpdate();
                        if (it.talla != null) {
                            stock.setInt(1, it.cantidad);
                            stock.setInt(2, it.producto.id);
                            stock.setString(3, it.talla);
                            stock.setInt(4, it.cantidad);
                            if (stock.executeUpdate() == 0) {
                                throw new SQLException("Stock insuficiente para " + it.producto.nombre
                                        + " talla " + it.talla);
                            }
                        }
                    }
                }
                con.commit();
            } catch (SQLException e) {
                con.rollback();
                throw e;
            } finally {
                con.setAutoCommit(true);
            }
        }

        List<Object[]> listarVentas() throws SQLException {
            List<Object[]> filas = new ArrayList<>();
            String sql = "SELECT v.id, v.fecha, v.cliente, v.metodo, v.total, "
                    + "(SELECT group_concat(nombre || COALESCE(' [' || talla || ']','') || ' x' || cantidad, ', ') "
                    + " FROM venta_items WHERE venta_id=v.id) "
                    + "FROM ventas v ORDER BY v.id DESC";
            try (Statement st = con.createStatement(); ResultSet rs = st.executeQuery(sql)) {
                while (rs.next()) {
                    filas.add(new Object[]{rs.getInt(1), rs.getString(2), rs.getString(3),
                            rs.getString(4), fmt(rs.getDouble(5)), rs.getString(6)});
                }
            }
            return filas;
        }
    }

    /** Busca un producto por código de barras exacto (null si no existe). */
    private Producto buscarPorCodigo(String codigo) {
        String c = codigo == null ? "" : codigo.trim();
        if (c.isEmpty()) return null;
        for (Producto p : productos) {
            if (p.codigoBarras != null && p.codigoBarras.equalsIgnoreCase(c)) return p;
        }
        return null;
    }

    /** Escanear producto: pide el código de barras (o el nombre exacto) y lo agrega al carrito. */
    private void escanear() {
        String s = JOptionPane.showInputDialog(this, "Escanea o escribe el código de barras:");
        if (s == null || s.trim().isEmpty()) return;
        Producto p = buscarPorCodigo(s);
        if (p == null) {
            for (Producto o : productos) {
                if (o.nombre.equalsIgnoreCase(s.trim())) { p = o; break; }
            }
        }
        if (p != null) agregarAlCarrito(p);
        else error("Producto no encontrado: " + s.trim());
    }

    private void refrescarProductos() {
        panelProductos.removeAll();
        panelProductos.add(new TarjetaProducto(null));

        String q = txtBuscar == null ? "" : txtBuscar.getText().trim().toLowerCase();
        for (Producto pr : productos) {
            if (!q.isEmpty() && !pr.nombre.toLowerCase().contains(q)
                    && !(pr.codigoBarras != null && pr.codigoBarras.toLowerCase().contains(q))) continue;
            if (filtro.equals("En descuento") && !pr.enDescuento) continue;
            if (filtro.equals("Destacado") && !pr.destacado) continue;
            panelProductos.add(new TarjetaProducto(pr));
        }
        int celdas = panelProductos.getComponentCount();
        while (celdas % 3 != 0) {
            JPanel vacio = new JPanel();
            vacio.setOpaque(false);
            vacio.setPreferredSize(new Dimension(140, 140));
            panelProductos.add(vacio);
            celdas++;
        }
        panelProductos.revalidate();
        panelProductos.repaint();
    }

    /** Tarjeta de producto con slot de imagen. producto == null => tarjeta "+" */
    class TarjetaProducto extends JPanel {
        private final Producto producto;

        TarjetaProducto(Producto producto) {
            this.producto = producto;
            setPreferredSize(new Dimension(140, 140));
            setCursor(new Cursor(Cursor.HAND_CURSOR));
            if (producto != null) {
                setToolTipText(producto.nombre + (producto.codigoBarras == null ? "" : "  |  Cód: " + producto.codigoBarras));
            }
            addMouseListener(new MouseAdapter() {
                @Override public void mouseClicked(MouseEvent e) {
                    if (SwingUtilities.isRightMouseButton(e) && producto != null) {
                        menuContextual(e);
                    } else if (SwingUtilities.isLeftMouseButton(e)) {
                        if (producto == null) editarProducto(null);
                        else agregarAlCarrito(producto);
                    }
                }
            });
        }

        private void menuContextual(MouseEvent e) {
            JPopupMenu menu = new JPopupMenu();
            JMenuItem editar = new JMenuItem("Editar producto (tallas / stock)...");
            editar.addActionListener(a -> editarProducto(producto));
            JMenuItem cambiar = new JMenuItem("Cambiar imagen...");
            cambiar.addActionListener(a -> {
                String ruta = elegirImagen();
                if (ruta != null) {
                    producto.setImagen(ruta);
                    try { bd.actualizarProducto(producto); }
                    catch (SQLException ex) { error("No se pudo guardar:\n" + ex.getMessage()); }
                    refrescarProductos();
                    refrescarCarrito();
                }
            });
            JMenuItem eliminar = new JMenuItem("Eliminar producto");
            eliminar.addActionListener(a -> {
                if (JOptionPane.showConfirmDialog(PuntoVenta.this,
                        "¿Eliminar \"" + producto.nombre + "\"?", "Eliminar",
                        JOptionPane.YES_NO_OPTION) != JOptionPane.YES_OPTION) return;
                try {
                    bd.eliminarProducto(producto);
                    productos.remove(producto);
                    carrito.removeIf(it -> it.producto == producto);
                    refrescarProductos();
                    refrescarCarrito();
                } catch (SQLException ex) { error("No se pudo eliminar:\n" + ex.getMessage()); }
            });
            menu.add(editar);
            menu.add(cambiar);
            menu.add(eliminar);
            menu.show(this, e.getX(), e.getY());
        }

        @Override protected void paintComponent(Graphics g0) {
            Graphics2D g = (Graphics2D) g0.create();
            g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
            int w = getWidth(), h = getHeight();

            if (producto == null) {
                g.setColor(new Color(190, 190, 190));
                g.fillRect(0, 0, w, h);
                g.setColor(Color.WHITE);
                g.setStroke(new BasicStroke(4));
                g.drawLine(w / 2 - 18, h / 2, w / 2 + 18, h / 2);
                g.drawLine(w / 2, h / 2 - 18, w / 2, h / 2 + 18);
            } else {
                g.setColor(Color.WHITE);
                g.fillRect(0, 0, w, h);
                if (producto.imagen != null) {
                    dibujarImagenAjustada(g, producto.imagen, 0, 0, w, h);
                } else {
                    g.setColor(new Color(245, 245, 245));
                    g.fillRect(0, 0, w, h);
                    g.setColor(new Color(170, 170, 170));
                    g.setFont(getFont().deriveFont(Font.BOLD, 13f));
                    String t = "SIN IMAGEN";
                    g.drawString(t, (w - g.getFontMetrics().stringWidth(t)) / 2, h / 2 - 10);
                }
                // Etiqueta de stock / agotado / descuento
                g.setFont(getFont().deriveFont(Font.BOLD, 10f));
                FontMetrics fm0 = g.getFontMetrics();
                if (producto.tieneTallas()) {
                    boolean agotado = producto.stockTotal() == 0;
                    String s = agotado ? "AGOTADO" : "Stock: " + producto.stockTotal();
                    g.setColor(agotado ? RED : TEAL);
                    g.fillRect(0, 0, fm0.stringWidth(s) + 12, 16);
                    g.setColor(Color.WHITE);
                    g.drawString(s, 6, 12);
                }
                if (producto.enDescuento) {
                    String s = "OFERTA";
                    int bw = fm0.stringWidth(s) + 12;
                    g.setColor(ORANGE);
                    g.fillRect(w - bw, 0, bw, 16);
                    g.setColor(Color.WHITE);
                    g.drawString(s, w - bw + 6, 12);
                }
                // Franja inferior
                int fh = 46;
                g.setColor(new Color(90, 90, 90, 215));
                g.fillRect(0, h - fh, w, fh);
                g.setColor(Color.WHITE);
                g.setFont(getFont().deriveFont(Font.BOLD, 11f));
                FontMetrics fm = g.getFontMetrics();
                String precio = fmt(producto.precio);
                g.drawString(precio, w - fm.stringWidth(precio) - 6, h - fh + 14);
                dibujarTextoEnVarias(g, producto.nombre, 6, h - fh + 14, w - 12 - fm.stringWidth(precio) - 6);
            }
            g.dispose();
        }

        private void dibujarTextoEnVarias(Graphics2D g, String texto, int x, int y, int ancho) {
            FontMetrics fm = g.getFontMetrics();
            String[] palabras = texto.split(" ");
            StringBuilder linea = new StringBuilder();
            int yy = y;
            for (String pal : palabras) {
                String prueba = linea.length() == 0 ? pal : linea + " " + pal;
                if (fm.stringWidth(prueba) > ancho && linea.length() > 0) {
                    g.drawString(linea.toString(), x, yy);
                    yy += fm.getHeight() - 2;
                    linea = new StringBuilder(pal);
                } else {
                    linea = new StringBuilder(prueba);
                }
            }
            g.drawString(linea.toString(), x, yy);
        }
    }

    private void agregarAlCarrito(Producto p) {
        String talla = null;
        if (p.tieneTallas()) {
            talla = elegirTalla(p);
            if (talla == null) return;
        }
        for (ItemCarrito it : carrito) {
            if (it.producto == p && Objects.equals(it.talla, talla)) {
                if (it.cantidad + 1 > it.stockDisponible()) { error("Stock insuficiente en talla " + talla); return; }
                it.cantidad++;
                refrescarCarrito();
                return;
            }
        }
        carrito.add(new ItemCarrito(p, talla));
        refrescarCarrito();
    }

    /** Muestra botones con las tallas que tienen stock. Devuelve la talla elegida o null. */
    private String elegirTalla(Producto p) {
        List<String> disp = new ArrayList<>();
        List<String> etiquetas = new ArrayList<>();
        for (Map.Entry<String, Integer> e : p.tallas.entrySet()) {
            int enCarrito = 0;
            for (ItemCarrito it : carrito) {
                if (it.producto == p && e.getKey().equals(it.talla)) enCarrito += it.cantidad;
            }
            int libre = e.getValue() - enCarrito;
            if (libre > 0) {
                disp.add(e.getKey());
                etiquetas.add(e.getKey() + "  (" + libre + ")");
            }
        }
        if (disp.isEmpty()) { error("Producto agotado"); return null; }
        int r = JOptionPane.showOptionDialog(this, "Elige la talla de " + p.nombre + "\n(entre paréntesis: stock)",
                "Talla", JOptionPane.DEFAULT_OPTION, JOptionPane.QUESTION_MESSAGE, null,
                etiquetas.toArray(), etiquetas.get(0));
        return r < 0 ? null : disp.get(r);
    }

    private void refrescarCarrito() {
        panelItems.removeAll();
        for (ItemCarrito it : carrito) panelItems.add(filaCarrito(it));
        panelItems.revalidate();
        panelItems.repaint();
        lblSubtotal.setText(fmt(subtotal()));
        lblTotal.setText(fmt(total()));
    }

    static class Miniatura extends JPanel {
        private final Producto p;
        Miniatura(Producto p) {
            this.p = p;
            setPreferredSize(new Dimension(44, 44));
            setOpaque(false);
        }
        @Override protected void paintComponent(Graphics g0) {
            Graphics2D g = (Graphics2D) g0.create();
            g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
            if (p.imagen != null) dibujarImagenAjustada(g, p.imagen, 0, 0, getWidth(), getHeight());
            else {
                g.setColor(new Color(235, 235, 235));
                g.fillRect(0, 0, getWidth(), getHeight());
                g.setColor(Color.GRAY);
                g.drawString("IMG", 11, 26);
            }
            g.dispose();
        }
    }

    private double subtotal() {
        double s = 0;
        for (ItemCarrito it : carrito) s += it.total();
        return s;
    }

    private double total() { return Math.max(0, subtotal() - descuento + envio); }

    private void limpiarVenta() {
        carrito.clear(); descuento = 0; envio = 0; nota = "";
        refrescarCarrito();
    }

    private void pagar() {
        if (carrito.isEmpty()) { error("El carrito está vacío"); return; }
        String[] metodos = {"Efectivo", "Tarjeta", "Transferencia"};
        Object m = JOptionPane.showInputDialog(this, "Total a pagar: " + fmt(total()) + "\nMétodo de pago:",
                "Pagar", JOptionPane.PLAIN_MESSAGE, null, metodos, metodos[0]);
        if (m == null) return;
        try {
            bd.registrarVenta(carrito, descuento, envio, total(), m.toString(),
                    txtCliente.getText().trim(), nota);
            // Reflejar el descuento de stock en memoria
            for (ItemCarrito it : carrito) {
                if (it.talla != null) it.producto.tallas.merge(it.talla, -it.cantidad, Integer::sum);
            }
            info("Venta registrada con " + m + " por " + fmt(total()));
            limpiarVenta();
            refrescarProductos();
        } catch (SQLException ex) {
            error("No se pudo registrar la venta:\n" + ex.getMessage());
        }
    }

    // CREAR / EDITAR PRODUCTO (imagen + tallas + stock)
    // ----------------------------------------------------------------
    private void editarProducto(Producto existente) {
        boolean nuevo = existente == null;
        Producto p = nuevo ? new Producto("", 0, null) : existente;

        JTextField nombre = new JTextField(p.nombre, 18);
        JTextField codigo = new JTextField(p.codigoBarras == null ? "" : p.codigoBarras);
        JTextField precio = new JTextField(nuevo ? "" : String.valueOf(p.precio));
        JCheckBox chkDest = new JCheckBox("Destacado", p.destacado);
        JCheckBox chkDesc = new JCheckBox("En descuento (oferta)", p.enDescuento);

        final String[] ruta = {p.rutaImagen};
        JLabel lblRuta = new JLabel(ruta[0] == null ? "(sin imagen)" : new File(ruta[0]).getName());
        JButton btnImg = new JButton("Elegir imagen...");
        btnImg.addActionListener(e -> {
            String r = elegirImagen();
            if (r != null) { ruta[0] = r; lblRuta.setText(new File(r).getName()); }
        });

        // Tallas: casilla + stock
        JCheckBox[] chk = new JCheckBox[TALLAS_STD.length];
        JSpinner[] spin = new JSpinner[TALLAS_STD.length];
        JPanel pTallas = new JPanel(new GridLayout(0, 2, 6, 4));
        pTallas.setBorder(BorderFactory.createTitledBorder("Tallas y stock (sin marcar = sin tallas)"));
        for (int i = 0; i < TALLAS_STD.length; i++) {
            Integer stock = p.tallas.get(TALLAS_STD[i]);
            chk[i] = new JCheckBox(TALLAS_STD[i], stock != null);
            spin[i] = new JSpinner(new SpinnerNumberModel(stock == null ? 10 : stock, 0, 99999, 1));
            spin[i].setEnabled(stock != null);
            final int idx = i;
            chk[i].addActionListener(e -> spin[idx].setEnabled(chk[idx].isSelected()));
            pTallas.add(chk[i]);
            pTallas.add(spin[i]);
        }

        JPanel datos = new JPanel(new GridLayout(0, 2, 6, 6));
        datos.add(new JLabel("Código de barras:")); datos.add(codigo);
        datos.add(new JLabel("Nombre:"));    datos.add(nombre);
        datos.add(new JLabel("Precio (S/):")); datos.add(precio);
        datos.add(btnImg);                   datos.add(lblRuta);
        datos.add(chkDest);                  datos.add(chkDesc);

        JPanel panel = new JPanel(new BorderLayout(0, 8));
        panel.add(datos, BorderLayout.NORTH);
        panel.add(pTallas, BorderLayout.CENTER);

        int r = JOptionPane.showConfirmDialog(this, panel, nuevo ? "Nuevo producto" : "Editar producto",
                JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);
        if (r != JOptionPane.OK_OPTION) return;

        double pr;
        try {
            pr = Double.parseDouble(precio.getText().replace(",", "."));
            if (nombre.getText().trim().isEmpty() || pr < 0) throw new NumberFormatException();
        } catch (NumberFormatException ex) {
            error("Nombre o precio inválido");
            return;
        }

        String cod = codigo.getText().trim();
        for (Producto o : productos) {
            if (o != p && !cod.isEmpty() && cod.equalsIgnoreCase(o.codigoBarras)) {
                error("Ese código de barras ya pertenece a: " + o.nombre);
                return;
            }
        }

        p.codigoBarras = cod.isEmpty() ? null : cod;
        p.nombre = nombre.getText().trim().toUpperCase();
        p.precio = pr;
        p.destacado = chkDest.isSelected();
        p.enDescuento = chkDesc.isSelected();
        if (!Objects.equals(ruta[0], p.rutaImagen)) p.setImagen(ruta[0]);
        p.tallas.clear();
        for (int i = 0; i < TALLAS_STD.length; i++) {
            if (chk[i].isSelected()) p.tallas.put(TALLAS_STD[i], (Integer) spin[i].getValue());
        }

        try {
            if (nuevo) { bd.insertarProducto(p); productos.add(p); }
            else bd.actualizarProducto(p);
        } catch (SQLException ex) {
            error("No se pudo guardar el producto:\n" + ex.getMessage());
            return;
        }
        sanearCarrito();
        refrescarProductos();
        refrescarCarrito();
    }

    /** Ajusta el carrito si cambió el stock o se quitaron tallas. */
    private void sanearCarrito() {
        carrito.removeIf(it -> it.talla != null && !it.producto.tallas.containsKey(it.talla));
        for (ItemCarrito it : carrito) {
            if (it.cantidad > it.stockDisponible()) it.cantidad = Math.max(1, it.stockDisponible());
        }
    }

    private String elegirImagen() {
        JFileChooser fc = new JFileChooser();
        fc.setFileFilter(new javax.swing.filechooser.FileNameExtensionFilter(
                "Imágenes (png, jpg, gif)", "png", "jpg", "jpeg", "gif"));
        if (fc.showOpenDialog(this) == JFileChooser.APPROVE_OPTION) {
            return fc.getSelectedFile().getAbsolutePath();
        }
        return null;
    }

    static void dibujarImagenAjustada(Graphics2D g, BufferedImage img, int x, int y, int w, int h) {
        double esc = Math.min((double) w / img.getWidth(), (double) h / img.getHeight());
        int iw = (int) (img.getWidth() * esc), ih = (int) (img.getHeight() * esc);
        g.drawImage(img, x + (w - iw) / 2, y + (h - ih) / 2, iw, ih, null);
    }

    public static void main(String[] args) {
        try { UIManager.setLookAndFeel(UIManager.getCrossPlatformLookAndFeelClassName()); }
        catch (Exception ignored) { }
        SwingUtilities.invokeLater(() -> {
            try {
                BD bd = new BD("puntoventa.db");
                new PuntoVenta(bd).setVisible(true);
            } catch (SQLException e) {
                JOptionPane.showMessageDialog(null,
                        "No se pudo abrir la base de datos.\n"
                        + "¿Agregaste el JAR sqlite-jdbc a las Libraries del proyecto?\n\n" + e.getMessage(),
                        "Error de base de datos", JOptionPane.ERROR_MESSAGE);
            }
        });
    }
